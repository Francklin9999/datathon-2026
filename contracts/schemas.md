# contracts/schemas.md — data types

Types are written in TypeScript notation because it is compact; the Pydantic models mirror them one-to-one (see `contracts/README.md`). Enumerations marked *(catalog)* are listed in `contracts/catalog.md`.

Notation: `?` = optional key; `| null` = key always present, value may be null; `Pct` = number in percent units (25.0 = 25%).

---

## 1. Identifiers

| ID | Format | Example | Notes |
|---|---|---|---|
| `ProfileId` | `robert` \| `amira` \| `mateo` for defaults; `p_` + 10 lowercase base32 chars for custom | `p_k3v9x2m7qa` | Defaults are read-only except rule thresholds, which are stored per session. |
| `SessionId` | `s_` + UUIDv4 | `s_6f1c…` | Generated by the browser, sent in `X-Aplomb-Session` (A-20). |
| `RunId` | `run_` + ULID | `run_01JCM9…` | Sortable by creation time. |
| `FactId` | `F-` + at least 3 digits, unique **within a run or check response** | `F-012` | Global reference = `runId` + `/` + `factId`. |
| `FactKey` | `metric|subjectType:subjectId|variant|window` | `vol.annualized|portfolio:robert|after|full` | Stable across recalculations; lets the UI match "the same number" between a run and a later `/check`. |
| `EvidenceId` | `E-` + at least 3 digits, unique within a run | `E-003` | |
| `RuleId` | dotted slug *(catalog)* | `cap.position` | Stable identity of a rule inside a profile. |
| `RuleCode` | `R` + position in the profile's rule list | `R1` | Display only; may change when rules are reordered. |
| `ScenarioId` | `sc_` + short slug | `sc_requested` | Unique within a run. |
| `EventId` | `evt_` + first 12 hex chars of SHA-1 of the canonical source URL | `evt_3fa9c1d2b7e0` | |
| `DatasetId` | snake_case *(catalog)* | `prices_daily` | |
| `DataVersion` | `snap-` + `YYYYMMDD` + `-` + label | `snap-20261113-starter` | One version covers all datasets in a snapshot. |
| `Ticker` | as in the source, upper case | `TSLA`, `XIU.TO`, `BTC-USD`, `CASH.CAD` | `CASH.CAD` / `CASH.USD` are pseudo-assets for cash. |

## 2. Primitives

```ts
type ISODate = string;        // "2026-11-12"
type ISODateTime = string;    // "2026-11-14T20:25:00Z"
type Currency = "CAD" | "USD";
type Lang = "en" | "fr";

interface Period { start: ISODate; end: ISODate; label?: string }   // inclusive on both ends

type WindowCode = "full" | "1y" | "3y" | "5y" | "ytd" | "custom";
interface WindowSpec { code: WindowCode; start?: ISODate; end?: ISODate }  // start/end required when code = "custom"

interface ApiError {
  error: { code: ErrorCode /* catalog */; message: string; details?: Record<string, unknown>; requestId: string }
}
```

## 3. Assets and universe

```ts
type AssetClass = "equity" | "etf_equity" | "etf_bond" | "etf_commodity" | "crypto" | "cash";
type AssetGroup = "equity" | "fixed_income" | "commodity" | "crypto" | "cash";
// Mapping: equity, etf_equity -> equity; etf_bond -> fixed_income; etf_commodity -> commodity; crypto -> crypto; cash -> cash.

interface Asset {
  ticker: Ticker;
  name: string;
  assetClass: AssetClass;
  assetGroup: AssetGroup;
  market: string;                 // as in assets_universe.csv, e.g. "TSX", "NASDAQ", "Crypto (24/7)"
  currency: Currency;             // listing currency
  sector: string;                 // label from the universe file (organizer-assigned for starter assets)
  gicsSector: string | null;      // same label when it is one of the 11 GICS sector names, else null (ETFs, crypto)
  isSingleName: boolean;          // true for assetClass equity and crypto; drives cap.position
  country: string | null;         // ISO-3166 alpha-3 of the listing market ("CAN", "USA"); null for crypto and cash
  themes: ThemeTag[];             // catalog; documented assumptions, see data/SPEC.md
  tier: "starter" | "extended" | "on_demand";
  history: { firstDate: ISODate; lastDate: ISODate; nObs: number; nObsMaster: number };
  flags: AssetFlag[];             // catalog: "short_history", "zero_volume_rows", "split_anomaly", ...
  dataVersion: DataVersion;
  datasetId: DatasetId;           // where its prices come from
}
```

## 4. Profiles, portfolios, rules

```ts
type RiskLevel = "low" | "low_moderate" | "moderate" | "moderate_high" | "high";
type Objective = "preserve" | "balanced" | "growth";
type Experience = "beginner" | "intermediate" | "advanced";

interface Preferences {
  themesOfInterest: ThemeTag[];
  excludedSectors: string[];            // sector labels
  excludedAssetClasses: AssetClass[];
  esgFocus: boolean;
  avoidControversies: boolean;
  avoidSpeculative: boolean;
  cryptoInterest: "none" | "low" | "open";
  wantsDiversification: boolean;
  notes: string;                         // free text kept verbatim, never interpreted by code
}

interface Position {
  ticker: Ticker;
  weightPct: number;                     // of total portfolio value; positions sum to 100 ± 0.01
}

interface Portfolio {
  isFictional: true;                     // always true; the UI prints it
  totalValue: number;                    // in `currency`
  currency: Currency;                    // = profile.reportingCurrency
  positions: Position[];
  updatedAt: ISODateTime;
}

interface Profile {
  profileId: ProfileId;
  isDefault: boolean;
  origin: "brief" | "user" | "agent_draft";
  name: string;
  initial: string;                       // one letter for the avatar
  age: number | null;
  occupation: string | null;
  riskLevel: RiskLevel;
  horizonYears: { min: number; max: number | null };   // max null = open-ended ("15 years or more")
  objective: Objective;
  objectiveText: I18nText;
  experience: Experience;
  reportingCurrency: Currency;
  preferences: Preferences;
  blurb: I18nText;                       // one or two sentences shown on the persona card
  personaTest: I18nText | null;          // "what this persona tests", defaults only
  portfolio: Portfolio;
  rules: PolicyRule[];
  version: number;                       // incremented on every change; used for optimistic locking
  createdAt: ISODateTime;
  updatedAt: ISODateTime;
}
```

### 4.1 PolicyRule

```ts
type RuleType = "cap" | "floor" | "limit" | "exclude" | "evidence" | "flag" | "sim";
type RuleOrigin = "derived" | "profile_preference" | "user" | "system";
type Comparator = "<=" | ">=" | "==" | "none";
type Unit = "pct" | "pts" | "ratio" | "count" | "days" | "cad" | "usd" | "date" | "text" | "bool";

interface PolicyRule {
  ruleId: RuleId;                        // catalog
  code: RuleCode;                        // "R1"...
  type: RuleType;
  enabled: boolean;
  label: I18nText;                       // plain-language sentence WITHOUT the number, with "{threshold}" placeholder
  rationale: I18nText;                   // why this rule exists for this profile, citing the profile text
  metric: MetricCode | null;             // catalog; what is measured
  target: { dimension: "position" | "sector" | "asset_class" | "asset_group" | "theme" | "portfolio" | "none"; key: string | null };
  comparator: Comparator;
  threshold: number | null;
  unit: Unit;
  editable: boolean;
  bounds: { min: number; max: number; step: number } | null;   // allowed range for edits
  origin: RuleOrigin;
  isAssumption: boolean;                 // true for everything except type "sim"; the UI shows an "assumption" tag
  derivation: {                          // present when origin = "derived"
    methodId: MethodId;                  // catalog, e.g. "M_DERIVE_POSITION_CAP"
    formula: string;                     // human-readable formula
    inputs: Record<string, number | string>;
    dataVersion: DataVersion;
    window: WindowSpec;
  } | null;
  countsInVerdict: boolean;              // false for type "flag" and "sim"
  updatedAt: ISODateTime;
}
```

Rule semantics, evaluation and the default rule sets are specified in `backend/aplomb/rules/SPEC.md`.

## 5. Requests: moves and goals

```ts
type FundingMode = "auto" | "pro_rata" | "from_cash" | "from_holdings" | "new_money";

interface Funding {
  mode: FundingMode;                     // "auto" = let the system compare and pick (D-15)
  fromTickers?: Ticker[];                // required when mode = "from_holdings"
}

interface ProposedMove {
  ticker: Ticker;
  sizePct: number;                       // > 0, points of CURRENT portfolio value added to `ticker` (A-17)
  funding: Funding;
}

interface ExposureGoal {
  themes: ThemeTag[];                    // at least one of themes / assetClasses / tickers must be non-empty
  assetClasses: AssetClass[];
  tickers: Ticker[];                     // assets the user named as examples
  targetExposurePct: number | null;      // desired exposure to the theme after the change; null = "as much as fits"
  maxLossPct: number | null;             // user's own stress limit for this request (overrides nothing; adds a constraint)
  maxVolPct: number | null;
  notes: string;                         // remaining free text
}

type DecisionRequest =
  | { type: "move"; move: ProposedMove }
  | { type: "goal"; goal: ExposureGoal };

interface ParsedRequest {                // response of POST /requests/parse
  request: DecisionRequest | null;       // null when nothing usable was understood
  interpretation: I18nText;              // one sentence restating the request for confirmation
  confidence: "high" | "medium" | "low";
  missing: { field: string; question: I18nText }[];
  warnings: I18nText[];                  // e.g. ticker not in the universe, will be fetched on demand
  refused: { reason: "trade_execution" | "advice_request" | "off_topic"; message: I18nText } | null;
}
```

## 6. Facts and evidence (the ledger)

```ts
type FactKind =
  | "observed"      // read directly from a dataset (a price, a rate)
  | "computed"      // calculated by code from observed data
  | "assumption"    // a threshold or parameter chosen by the team or the user
  | "input"         // something the user typed (move size, as-of date)
  | "scenario"      // result under a hypothetical (stress replay, alternative move)
  | "rule_check"    // outcome of a rule evaluation
  | "evidence";     // a figure quoted from a cited document

interface SourceRef {
  datasetId: DatasetId;
  dataVersion: DataVersion;
  sourceName: string;                    // "Yahoo Finance via yfinance", "Bank of Canada Valet API"
  sourceUrl: string | null;
  retrievedOn: ISODate;
  licence: string;                       // short text or "see passport"
  rowsRead: number | null;
}

interface Fact {
  id: FactId;
  key: FactKey;
  kind: FactKind;
  metric: MetricCode;                    // catalog
  label: I18nText;                       // "Tesla (TSLA) return, calendar 2022"
  value: number | string | boolean | null;
  unit: Unit;
  currency: Currency | null;
  display: I18nText;                     // formatted by the backend: "−65.0%" / "−65,0 %"
  subject: {
    type: "asset" | "portfolio" | "sector" | "asset_group" | "asset_class" | "theme" | "country" | "rule" | "macro" | "event" | "request" | "universe";
    id: string;                          // ticker, profileId, sector label, ruleId, series id...
    variant: "before" | "after" | "na" | string;   // "after", or "scenario:<ScenarioId>"
  };
  period: Period | null;
  window: WindowSpec | null;
  asOf: ISODate;                         // information cut-off used for this fact
  availableAt: ISODateTime | null;       // latest availability timestamp among the rows used
  method: { id: MethodId; description: I18nText; params: Record<string, number | string | boolean> };
  inputs: FactId[];                      // facts this one was computed from (same run)
  nObs: number | null;
  sources: SourceRef[];
  limitations: LimitationCode[];         // catalog
  verification: {
    status: "verified" | "unverified" | "failed";
    checks: { code: string; ok: boolean; detail: string }[];
  };
  createdBy: "quant" | "rules" | "optimizer" | "research" | "profile" | "user" | "world" | "timemachine";
  createdAt: ISODateTime;
}

type EvidenceType = "filing" | "company_report" | "fund_document" | "official_release" | "news" | "registry" | "attention" | "web_page";

interface Evidence {
  id: EvidenceId;
  type: EvidenceType;
  tier: 1 | 2 | 3;                       // A-25
  title: string;
  publisher: string;
  url: string;
  publishedAt: ISODate | null;
  retrievedAt: ISODateTime;
  lang: string;                          // BCP-47 of the source document
  quote: string | null;                  // verbatim, at most 400 characters
  claim: I18nText;                       // what this evidence is used to support, in our words
  subject: { type: "asset" | "sector" | "country" | "event" | "theme"; id: string };
  tool: string;                          // tool name from contracts/tools.md
  documentRef: string | null;            // storage key of the cached document text used for quote verification
  licence: string;                       // "public domain (US gov)", "Open Government Licence – Canada", "unknown"...
  verification: {
    status: "verified" | "unverified" | "failed";
    checks: { code: string; ok: boolean; detail: string }[];   // e.g. A08 quote found in cached document
  };
}
```

## 7. Rule checks, verdicts, scenarios

```ts
type RuleStatus = "pass" | "tension" | "conflict" | "nee" | "info";   // nee = not enough evidence
type Verdict = "compatible" | "tension" | "conflict" | "insufficient";

interface RuleCheck {
  ruleId: RuleId;
  code: RuleCode;
  status: RuleStatus;
  value: number | null;                  // measured value, in rule.unit
  threshold: number | null;
  unit: Unit;
  comparator: Comparator;
  headroom: number | null;               // threshold − value for caps/limits, value − threshold for floors; null otherwise
  worst: { type: "asset" | "sector" | "asset_group" | "theme"; id: string } | null;   // which item drives the value
  valueFactId: FactId | null;
  thresholdFactId: FactId | null;
  detail: I18nText;                      // one sentence built by code from a template
  neeReason: NeeReason | null;           // catalog: "no_holdings", "short_history", "no_evidence_loaded", ...
  evidenceIds: EvidenceId[];
}

interface VerdictCounts { pass: number; tension: number; conflict: number; nee: number; info: number }

interface PortfolioSnapshot {            // a portfolio with derived figures, before or after a move
  variant: "before" | "after" | string;
  totalValue: number;
  currency: Currency;
  positions: (Position & { name: string; assetClass: AssetClass; assetGroup: AssetGroup; sector: string; marketValue: number })[];
  byAssetGroup: { key: AssetGroup; weightPct: number }[];
  bySector: { key: string; weightPct: number }[];          // direct single-name equities only
  byCountry: { key: string; weightPct: number }[];
  metrics: {
    volatilityPct: number | null; maxDrawdownPct: number | null; stress2022Pct: number | null;
    factIds: Record<string, FactId>;     // metric name -> fact
  };
}

interface FundingOption {
  funding: Funding;                      // resolved: never "auto"
  feasible: boolean;
  infeasibleReason: I18nText | null;
  verdict: Verdict | null;
  counts: VerdictCounts | null;
  maxCompatibleSizePct: number | null;
  recommended: boolean;                  // exactly one feasible option has true
  rank: number | null;
  note: I18nText;
}

type ScenarioKind = "requested" | "max_compatible" | "alternative" | "goal_candidate" | "rebalance";

interface Scenario {
  scenarioId: ScenarioId;
  kind: ScenarioKind;
  label: I18nText;
  moves: ProposedMove[];                 // one for a simple move; several for a rebalance
  verdict: Verdict;
  counts: VerdictCounts;
  ruleChecks: RuleCheck[];
  portfolioAfter: PortfolioSnapshot;
  exposure: { theme: ThemeTag; beforePct: number; afterPct: number }[];   // for goal requests
  rationale: I18nText;                   // why this scenario was built (by code template or writer tokens)
  isBestFit: boolean;
  rank: number;
  keyFactIds: FactId[];
}

interface CheckResult {                  // response of POST /check (deterministic, no model)
  verdict: Verdict;
  counts: VerdictCounts;
  ruleChecks: RuleCheck[];
  portfolioBefore: PortfolioSnapshot;
  portfolioAfter: PortfolioSnapshot;
  fundingUsed: Funding;                  // resolved
  fundingOptions: FundingOption[];       // filled when the request asked for "auto" or compareFunding = true
  maxCompatibleSizePct: number;
  baselineConflict: boolean;             // true when the portfolio already conflicts before the move
  facts: Record<FactId, Fact>;
  asOf: ISODate;
  dataVersion: DataVersion;
  window: WindowSpec;
  computeMs: number;
}
```

Verdict rule (implemented once, in `rules`): `conflict` if any counted rule is in conflict; else `tension` if any is in tension; else `insufficient` if any is `nee`; else `compatible`. Counts include all returned statuses, including `info`. `maxCompatibleSizePct` in checks, funding options and decision results is the first-conflict grid boundary; tension and evidence gaps can remain at that size.

## 8. Brief (written explanation) and audit

```ts
type Segment =
  | { t: "text"; text: string }
  | { t: "fact"; factId: FactId; display: string }            // value inserted by code
  | { t: "rule"; ruleId: RuleId; display: string }            // rule code + short label
  | { t: "status"; ruleId: RuleId; display: string }          // status word inserted by code
  | { t: "evidence"; evidenceId: EvidenceId; display: string };

interface Sentence {
  id: string;                            // "S-01"
  segments: Segment[];
  audit: { status: "verified" | "struck" | "repaired"; failedChecks: AuditCheckCode[]; note: I18nText | null };
}

type BriefSectionId = "summary" | "facts" | "assumptions" | "limits" | "scenarios" | "next_checks";

interface BriefSection { id: BriefSectionId; title: I18nText; sentences: Sentence[] }

interface AuditCheckResult { code: AuditCheckCode; ok: boolean; kind: "code" | "llm" | "guardrail"; detail: string; sentenceIds: string[] }

interface AuditReport {
  status: "passed" | "passed_with_strikes" | "failed";
  checks: AuditCheckResult[];
  sentencesChecked: number;
  sentencesStruck: number;
  rewrites: number;                      // 0 or 1
  guardrail: { applied: boolean; action: "none" | "blocked" | "masked"; id: string | null };
  llmReview: { model: string | null; findings: { sentenceId: string; issue: string; code: AuditCheckCode }[] };
  ledgerHash: string;                    // SHA-256 over the canonical JSON of facts + evidence
  signedAt: ISODateTime;
}

interface Brief {
  lang: Lang;
  sections: BriefSection[];              // "facts", "assumptions", "limits" are mandatory
  disclaimer: I18nText;                  // simulation-only text, inserted by code
  generatedAt: ISODateTime;
  model: string;                         // Bedrock model ID used by the writer
  audit: AuditReport;
}
```

## 9. Runs

```ts
type RunKind = "decision" | "portfolio_brief" | "event_brief" | "replay" | "profile_draft" | "sabotage";
type RunStatus = "queued" | "running" | "waiting_input" | "succeeded" | "failed" | "cancelled";
type StepType = "plan" | "research" | "quant" | "risk_check" | "revise" | "write" | "audit" | "sign";
type AgentRole = "planner" | "researcher" | "quant" | "risk_officer" | "writer" | "auditor";

interface ToolCall {
  id: string;                            // "T-001"
  tool: string;                          // contracts/tools.md
  args: Record<string, unknown>;         // as sent, secrets never included
  startedAt: ISODateTime;
  durationMs: number;
  ok: boolean;
  resultSummary: string;                 // short, human-readable
  datasetsRead: { datasetId: DatasetId; dataVersion: DataVersion; rows: number; rowsWithheld: number }[];
  factIds: FactId[];
  evidenceIds: EvidenceId[];
  error: string | null;
}

interface Step {
  seq: number;                           // 1-based, strictly increasing within a run
  type: StepType;
  iteration: number;                     // 1 for the first pass; 2+ after a revise loop
  agent: AgentRole;
  actor: "llm" | "llm+tools" | "code" | "code+llm";
  status: "running" | "done" | "failed";
  startedAt: ISODateTime;
  endedAt: ISODateTime | null;
  durationMs: number | null;
  summary: I18nText;                     // what was done, built from templates + counts
  model: string | null;                  // Bedrock model ID when an LLM was called
  usage: { inputTokens: number; outputTokens: number } | null;
  toolCalls: ToolCall[];
  factIds: FactId[];
  evidenceIds: EvidenceId[];
}

interface ClarifyingQuestion {
  questionId: string;
  text: I18nText;
  field: string;                         // which input is missing
  options: { value: string; label: I18nText }[] | null;   // null = free text
}

interface Run {
  runId: RunId;
  kind: RunKind;
  status: RunStatus;
  sessionId: SessionId;
  profileId: ProfileId | null;
  parentRunId: RunId | null;             // set for sabotage runs and for reruns
  lang: Lang;
  asOf: ISODate;                         // information cut-off (today's data date, or the Time Machine date)
  dataVersion: DataVersion;
  input: RunInput;
  steps: Step[];
  facts: Record<FactId, Fact>;
  evidence: Record<EvidenceId, Evidence>;
  question: ClarifyingQuestion | null;   // set when status = "waiting_input"
  result: RunResult | null;              // set when status = "succeeded"
  error: ApiError["error"] | null;       // set when status = "failed"
  flags: { sabotage: SabotageMode | null; fixture: boolean; timeMachine: boolean };
  usage: { llmCalls: number; inputTokens: number; outputTokens: number; toolCalls: number; durationMs: number };
  createdAt: ISODateTime;
  updatedAt: ISODateTime;
}

type SabotageMode = "invented_number" | "false_comparison" | "unsupported_interpretation";

type RunInput =
  | { kind: "decision"; request: DecisionRequest; originalText: string | null; window: WindowSpec }
  | { kind: "portfolio_brief"; window: WindowSpec }
  | { kind: "event_brief"; eventId: EventId }
  | { kind: "replay"; request: DecisionRequest; asOf: ISODate; window: WindowSpec }
  | { kind: "profile_draft"; text: string; fields: Partial<Profile> }
  | { kind: "sabotage"; mode: SabotageMode };

type RunResult =
  | { kind: "decision"; decision: DecisionResult }
  | { kind: "portfolio_brief"; portfolioBrief: PortfolioBriefResult }
  | { kind: "event_brief"; eventBrief: EventBriefResult }
  | { kind: "replay"; replay: ReplayResult }
  | { kind: "profile_draft"; draft: ProfileDraftResult }
  | { kind: "sabotage"; sabotage: SabotageResult };
```

### 9.1 Results by kind

```ts
interface DecisionResult {
  request: DecisionRequest;              // as resolved (funding never "auto")
  verdict: Verdict;                      // verdict of the selected scenario
  counts: VerdictCounts;
  selectedScenarioId: ScenarioId;
  scenarios: Scenario[];                 // always contains kind "requested" for a move; goal requests contain goal_candidate(s)
  portfolioBefore: PortfolioSnapshot;
  maxCompatibleSizePct: number | null;   // for move requests
  fundingOptions: FundingOption[];
  attention: AttentionSignal | null;     // hype meter, when available
  macroContext: FactId[];                // facts describing the rate/inflation backdrop

  limitations: LimitationCode[];

interface AttentionSignal {
  ticker: Ticker;
  article: string;                       // Wikipedia article title used
  ratio: number | null;                  // recent average views ÷ baseline average views
  recentPeriod: Period; baselinePeriod: Period;
  flag: "elevated" | "normal" | "unknown";
  factId: FactId | null;
  evidenceId: EvidenceId | null;
}

interface PortfolioBriefResult {
  snapshot: PortfolioSnapshot;           // variant "before"
  ruleChecks: RuleCheck[]; verdict: Verdict; counts: VerdictCounts;
  focusCards: { title: I18nText; body: Sentence[]; factIds: FactId[] }[];   // exactly 3
  brief: Brief;
  horizonBand: "short" | "medium" | "long";    // 0–3, 3–10, 10+ years (plan §5; product default)
}

interface ProfileDraftResult {

}

interface SabotageResult {
  mode: SabotageMode;
  injectedInstruction: string;           // the exact extra instruction given to the writer
  briefBeforeAudit: Brief;               // the lie visible, audit status "struck" on the offending sentence
  briefAfterAudit: Brief;
  caught: boolean;                       // true when at least one check failed on the injected sentence
  failedChecks: AuditCheckResult[];
}
```

## 10. World Explorer

```ts
type EventCategory = "monetary_policy" | "macro_release" | "geopolitical" | "trade_regulation" | "corporate"
                   | "energy_commodities" | "markets" | "natural_disaster" | "other";

interface EventSource { publisher: string; title: string; url: string; publishedAt: ISODateTime | null; sourceType: "news" | "official" | "filing" | "data"; lang: string }

interface WorldEvent {
  eventId: EventId;
  title: string;                         // headline or official title, verbatim from the source
  category: EventCategory;
  occurredAt: ISODate;
  publishedAt: ISODateTime | null;
```
